# Ava voice extras — optional. Enables STT (faster-whisper) and the
# speaker-verification voice gate (torch + speechbrain, CPU inference).
# Without these the bridge runs voice-less; /api/talk returns 503.
#
# TTS: AVA_TTS defaults to `auto`, which picks the best engine actually present —
# the kokoro sidecar if it answers, else Piper, else macOS `say` (built in, no
# install), else espeak-ng. Nothing here is required for speech on a Mac.
# For Piper, EITHER install the wheel (`pip install piper-tts`, which carries its
# own ONNX runtime) OR drop a platform build in bin/piper/ — that directory is
# gitignored, so it exists only on machines where someone put it there. It used
# to be the default engine, which meant every fork's default pointed at a file
# it did not have.
#
# Install:  pip install -r requirements-voice.txt
# Tip: for CPU-only torch wheels add:  --extra-index-url https://download.pytorch.org/whl/cpu
faster-whisper==1.2.1
torch==2.13.0
torchaudio==2.11.0
